"""Bounded source-only fixture evidence; never an environment-key allocator.

Scope coverage and exact nominal constructions are different from runtime-flow
coverage. Synthetic exclusion-range tests are kept separate from literal key
fixtures; they never establish that an entire range was environmentally used.
"""
from __future__ import annotations
import ast
import hashlib
import json
from pathlib import Path
import stat
import sys

MAX_ITEMS = 16384
MAX_NODES = 262144
MAX_BYTES = 1024 * 1024
MODULUS = 1 << 64
SINGLES = frozenset({"seed", "experiment_seed", "scenario_seed", "algorithm_seed",
    "last_visible_seed", "partial_seed", "seed_first", "seed_last",
    "base_seed", "seed_base", "seed_offset", "pilot_seed_base",
    "confirmatory_seed_base", "calibration_seed_offset"})
LISTS = frozenset({"seeds", "scientific_seeds", "spent_smoke_seeds",
    "fresh_smoke_seeds", "development_seeds", "stage_seeds", "smoke_seeds",
    "development_fresh_seeds", "ordered_seeds", "primary_seeds",
    "secondary_seeds", "engineering_keys", "seed_ids", "SEEDS", "KEYS",
    "STAGE_SEEDS", "primary_blocks", "secondary_blocks", "old_scientific"})
RANGES = frozenset({"old_engineering_ranges", "operational_fixture_ranges"})
FOLDER = Path(__file__).resolve().parent
ROOT = FOLDER.parents[3]
CONTRACT = FOLDER / "fixture_contract_candidate_v19.json"


class Unresolved(ValueError):
    pass


def _int(value):
    if type(value) is not int or value.bit_length() > 256:
        raise Unresolved("not-bounded-native-int")
    return value


def _bounded(items):
    if len(items) > MAX_ITEMS:
        raise Unresolved("item-budget")
    return items


def _expression(node, names, depth=0):
    """Closed arithmetic/list grammar only; no eval/import/callback/attribute."""
    if depth > 32:
        raise Unresolved("expression-depth")
    go = lambda n, env=names: _expression(n, env, depth + 1)
    if isinstance(node, ast.Constant):
        return _int(node.value)
    if isinstance(node, ast.Name):
        if node.id not in names:
            raise Unresolved("unknown-name:" + node.id)
        return names[node.id]
    if isinstance(node, ast.UnaryOp):
        x = _int(go(node.operand))
        if isinstance(node.op, ast.USub): return -x
        if isinstance(node.op, ast.UAdd): return x
    if isinstance(node, ast.BinOp):
        a, b = _int(go(node.left)), _int(go(node.right))
        if isinstance(node.op, ast.Add): return _int(a + b)
        if isinstance(node.op, ast.Sub): return _int(a - b)
        if isinstance(node.op, ast.Mult): return _int(a * b)
        if isinstance(node.op, ast.LShift) and 0 <= b <= 255:
            return _int(a << b)
    if isinstance(node, (ast.List, ast.Tuple)):
        return _bounded([go(n) for n in node.elts])
    if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and not node.keywords:
        # Local/module masking of these names is handled by context.
        name = node.func.id
        if name in names:
            raise Unresolved("shadowed-builtin")
        args = [go(n) for n in node.args]
        if name == "range" and 1 <= len(args) <= 3:
            r = range(*[_int(n) for n in args])
            if len(r) > MAX_ITEMS: raise Unresolved("range-budget")
            return list(r)
        if name in ("list", "tuple") and len(args) == 1 and type(args[0]) is list:
            return _bounded(list(args[0]))
    if isinstance(node, (ast.ListComp, ast.GeneratorExp)):
        values = []
        def expand(k, env):
            if k == len(node.generators):
                values.append(_expression(node.elt, env, depth + 1))
                _bounded(values)
                return
            g = node.generators[k]
            if not isinstance(g.target, ast.Name) or g.ifs or g.is_async:
                raise Unresolved("comprehension-target-filter-or-async")
            seq = _expression(g.iter, env, depth + 1)
            if type(seq) is not list: raise Unresolved("comprehension-iterator")
            for value in seq:
                branch = dict(env); branch[g.target.id] = value
                expand(k + 1, branch)
        if len(node.generators) > 3: raise Unresolved("generator-budget")
        expand(0, names)
        return values
    raise Unresolved("unsupported-expression:" + type(node).__name__)


def _flatten(value, depth=0):
    if depth > 32: raise Unresolved("value-depth")
    if type(value) is int: return [_int(value)]
    if type(value) is list:
        result = []
        for x in value:
            result.extend(_flatten(x, depth + 1)); _bounded(result)
        return result
    raise Unresolved("not-native-integer-container")


def _union(rows):
    answer = []
    for a, b in sorted(rows):
        if type(a) is not int or type(b) is not int or not (0 <= a <= b < MODULUS):
            raise Unresolved("invalid-canonical-interval")
        if answer and a <= answer[-1][1] + 1:
            answer[-1][1] = max(answer[-1][1], b)
        else: answer.append([a, b])
    return [[str(a), str(b)] for a, b in answer]


def inspect_fixture_python_bytes(blob):
    """Read an entire native bytes source; return declarations and explicit gaps.

    A declared single offset is an exclusion label, not expansion of its dynamic
    runtime schedule. No strings/bytes snippets are interpreted as actual run
    records. Their literal presence is counted as an unresolved parser-input
    coverage class rather than silently salvaging a malformed prefix.
    """
    if type(blob) is not bytes: raise TypeError("native bytes required")
    def refused(reason):
        return {"parse_complete": False, "claims": [], "fixture_alias_intervals": [],
            "synthetic_range_envelopes": [], "gaps": [{"line": 0, "reason": reason}],
            "runtime_key_use_proved": False, "runtime_flows_complete": False}
    if len(blob) > MAX_BYTES: return refused("whole-source-byte-budget")
    try:
        tree = ast.parse(blob.decode("utf-8-sig"))
        nodes = list(ast.walk(tree))
        if len(nodes) > MAX_NODES: return refused("whole-source-node-budget")
    except (SyntaxError, UnicodeError, RecursionError, MemoryError, ValueError) as exc:
        return refused("whole-source-refused:" + type(exc).__name__)
    parents = {child: parent for parent in nodes for child in ast.iter_child_nodes(parent)}
    claims, gaps, envelopes, aliases = [], [], [], []
    module_names = {}
    # Only unique module straight-line definitions. Any Store/Del masks a name,
    # apart from this single exact target; imports/parameters mask builtins too.
    def stores(scope):
        found = {}
        # Nested lexical scopes and comprehension targets do not bind the caller.
        def visit(n):
            if n is not scope and isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef,
                ast.ClassDef, ast.Lambda)):
                if hasattr(n, "name"): found.setdefault(n.name, []).append(n)
                return
            if isinstance(n, (ast.ListComp, ast.GeneratorExp, ast.SetComp, ast.DictComp)):
                return
            if isinstance(n, ast.Name) and isinstance(n.ctx, (ast.Store, ast.Del)):
                found.setdefault(n.id, []).append(n)
            if isinstance(n, (ast.Import, ast.ImportFrom)):
                for alias in n.names:
                    name = alias.asname or alias.name.split(".")[0]
                    found.setdefault(name, []).append(n)
            for c in ast.iter_child_nodes(n): visit(c)
        visit(scope)
        if isinstance(scope, (ast.FunctionDef, ast.AsyncFunctionDef)):
            for a in (*scope.args.posonlyargs, *scope.args.args, *scope.args.kwonlyargs):
                found.setdefault(a.arg, []).append(a)
            for a in (scope.args.vararg, scope.args.kwarg):
                if a is not None: found.setdefault(a.arg, []).append(a)
        return found
    module_stores = stores(tree)
    for stmt in tree.body:
        if isinstance(stmt, ast.Assign) and len(stmt.targets) == 1 and isinstance(stmt.targets[0], ast.Name):
            target = stmt.targets[0]
            if module_stores.get(target.id) == [target]:
                try: module_names[target.id] = _expression(stmt.value, module_names)
                except (Unresolved, ValueError, OverflowError): pass
    builtin_shadows = {"range", "list", "tuple"} & set(module_stores)
    for name in builtin_shadows: module_names[name] = "SHADOWED"
    def ancestors(node):
        out = []; current = node
        while current in parents:
            current = parents[current]; out.append(current)
        return out
    def environments(node):
        chain = ancestors(node)
        funcs = [x for x in chain if isinstance(x, (ast.FunctionDef, ast.AsyncFunctionDef))]
        if len(funcs) > 1: raise Unresolved("nested-closure-not-resolved")
        env = dict(module_names)
        if funcs:
            f = funcs[0]; local = stores(f)
            for name in local: env.pop(name, None)
            for name in {"range", "list", "tuple"} & set(local): env[name] = "SHADOWED"
            for stmt in f.body:
                if stmt.lineno >= node.lineno: break
                if isinstance(stmt, ast.Assign) and len(stmt.targets) == 1 and isinstance(stmt.targets[0], ast.Name):
                    target = stmt.targets[0]
                    if local.get(target.id) == [target]:
                        try: env[target.id] = _expression(stmt.value, env)
                        except (Unresolved, ValueError, OverflowError): pass
                elif not isinstance(stmt, (ast.Expr, ast.Pass)):
                    # No following assignment is certified as dominating through
                    # branches, loops, with/try, return, raise, or control transfer.
                    break
        envs = [env]
        for loop in reversed(chain):
            if not isinstance(loop, ast.For): continue
            body_nodes = {n for stmt in loop.body for n in ast.walk(stmt)}
            if node not in body_nodes or not isinstance(loop.target, ast.Name):
                raise Unresolved("loop-binding-not-in-simple-body")
            owner = funcs[0] if funcs else tree
            bindings = stores(owner).get(loop.target.id, [])
            if bindings != [loop.target]: raise Unresolved("mutated-loop-variable")
            next_envs = []
            for e in envs:
                seq = _expression(loop.iter, e)
                if type(seq) is not list: raise Unresolved("unknown-loop-iterator")
                for value in seq:
                    q = dict(e); q[loop.target.id] = value; next_envs.append(q)
                    _bounded(next_envs)
            envs = next_envs
        return envs
    def gap(node, reason):
        gaps.append({"line": node.lineno, "reason": reason})
    def record(node, field, expression, role="DECLARED_LITERAL_FIXTURE"):
        try:
            values = []
            for e in environments(node):
                raw = _expression(expression, e)
                if field in LISTS and type(raw) is int:
                    raise Unresolved("scalar-container-is-not-identity")
                values.extend(_flatten(raw)); _bounded(values)
            if field in RANGES:
                # A synthetic exclusion envelope never becomes a fixture-key union.
                envs = environments(node)
                for e in envs:
                    raw = _expression(expression, e)
                    if type(raw) is not list: raise Unresolved("range-container")
                    rows = []
                    for row in raw:
                        if type(row) is not list or len(row) != 2:
                            raise Unresolved("range-pair-shape")
                        a, b = map(_int, row)
                        if not 0 <= a <= b < MODULUS: raise Unresolved("range-bounds")
                        rows.append([str(a), str(b)])
                    envelopes.append({"line": node.lineno, "field": field,
                        "role": "SYNTHETIC_EXCLUSION_ENVELOPE_NOT_KEY_USE",
                        "intervals_inclusive_decimal": rows})
                return
            if values:
                originals = sorted(set(values))
                canonical = sorted({x % MODULUS for x in originals})
                aliases.extend((x, x) for x in canonical)
                claims.append({"line": node.lineno, "field": field, "role": role,
                    "expression": ast.get_source_segment(blob.decode("utf-8-sig"), expression),
                    "literal_occurrences": len(values),
                    "original_integers_decimal": [str(x) for x in originals],
                    "modulo64_aliases_decimal": [str(x) for x in canonical],
                    "environment_key_use_proved": False})
        except (Unresolved, ValueError, OverflowError, RecursionError) as exc:
            gap(node, str(exc))
    for node in nodes:
        if isinstance(node, ast.Dict):
            for k, v in zip(node.keys, node.values):
                if isinstance(k, ast.Constant) and type(k.value) is str and k.value in SINGLES | LISTS | RANGES:
                    record(node, k.value, v)
        elif isinstance(node, (ast.Assign, ast.AnnAssign, ast.AugAssign)):
            targets = node.targets if isinstance(node, ast.Assign) else [node.target]
            for target in targets:
                field = target.id if isinstance(target, ast.Name) else None
                if isinstance(target, ast.Subscript) and isinstance(target.slice, ast.Constant) and type(target.slice.value) is str:
                    field = target.slice.value
                if field in SINGLES | LISTS | RANGES:
                    if isinstance(node, ast.AugAssign):
                        gap(node, "augmented-mutation-not-a-complete-value")
                        if field in RANGES:
                            record(node, field, node.value)
                    else: record(node, field, node.value)
        elif isinstance(node, ast.keyword) and node.arg in SINGLES | LISTS:
            record(node, node.arg, node.value)
        elif isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute) and node.func.attr == "refuse_key":
            if node.args: record(node, "seed", node.args[0], "REJECTED_INPUT_FIXTURE_LABEL_NOT_ENVIRONMENT_USE")
        elif isinstance(node, ast.Constant) and type(node.value) in (str, bytes):
            text = node.value.decode("utf-8", errors="replace") if type(node.value) is bytes else node.value
            # Literal parser payloads are separate from ambient Python declarations.
            if ("seed" in text.casefold() or "KEYS" in text) and text.lstrip().startswith(("{", "[", "BASE", "seed", "for ", "import ")):
                gap(node, "embedded-parser-input-not-decoded-as-runtime-key-record")
    return {"parse_complete": True,
        "claims": sorted(claims, key=lambda r: (r["line"], r["field"], r["role"])),
        "fixture_alias_intervals": _union(aliases),
        "synthetic_range_envelopes": sorted(envelopes, key=lambda r: (r["line"], r["field"])),
        "gaps": sorted(gaps, key=lambda r: (r["line"], r["reason"])),
        "runtime_key_use_proved": False, "runtime_flows_complete": False}


def snapshot_preparation_fixtures():
    """Fixed current-parent 19 source files only. No scope/key/output override."""
    import readiness
    def sha(blob): return hashlib.sha256(blob).hexdigest()
    core_paths = (Path(__file__), CONTRACT, FOLDER / "test_fixture_source_candidate.py")
    core_before = {p.name: sha(p.read_bytes()) for p in core_paths}
    contract = json.loads(CONTRACT.read_text(encoding="utf-8"))
    expected = contract["bound_parent_test_sources"]
    readiness.require(len(expected) == 19, "fixed nineteen parent test sources")
    parent = readiness.load_parent(ROOT)
    readiness.require(parent["sources_digest"] == contract["parent"]["sources_digest"],
                      "parent source identity")
    readiness.require(parent["lock_sha256"] == contract["parent"]["lock_sha256"], "F1 identity")
    records, all_aliases, total_bytes = [], [], 0
    def read_bound(relative, wanted):
        p = ROOT / relative
        readiness.require(p.resolve().parent == FOLDER.resolve(), "exact source directory")
        st = p.lstat()
        readiness.require(stat.S_ISREG(st.st_mode) and not p.is_symlink() and
            not (getattr(st, "st_file_attributes", 0) & 0x400), "regular nonreparse source")
        raw = p.read_bytes()
        readiness.require(sha(raw) == wanted, "fixture source SHA mismatch:" + relative)
        return raw
    for name, wanted in sorted(expected.items()):
        raw = read_bound(name, wanted); total_bytes += len(raw)
        result = inspect_fixture_python_bytes(raw)
        readiness.require(result["parse_complete"], "entire fixed source parse")
        records.append({"path": name, "actual_bytes": len(raw), "sha256": sha(raw),
                        "source_only_inspection": result})
        all_aliases.extend((int(a), int(b)) for a, b in result["fixture_alias_intervals"])
    # Hash entire scope again, not just size/mtime/existence.
    for name, wanted in sorted(expected.items()): read_bound(name, wanted)
    after = {p.name: sha(p.read_bytes()) for p in core_paths}
    readiness.require(core_before == after, "candidate bytes moved during snapshot")
    parent_after = readiness.load_parent(ROOT)
    readiness.require(parent["sources_digest"] == parent_after["sources_digest"] and
        parent["lock_sha256"] == parent_after["lock_sha256"], "frozen source or F1 moved")
    payload = {"schema": "a96-B2-preparation-fixture-evidence-v19",
        "role": "SOURCE_ONLY_CONSERVATIVE_EXCLUSION_SUPPLEMENT_NOT_COLLECTION_GATE",
        "parent_commit": contract["parent"]["local_preparation_parent_commit"],
        "parent_sources_digest": parent["sources_digest"],
        "lock_sha256": parent["lock_sha256"], "core_before_after_sha256": core_before,
        "bound_scope_files": len(records), "actual_scope_bytes": total_bytes,
        "records": records, "fixture_alias_intervals_inclusive_decimal": _union(all_aliases),
        "synthetic_envelopes_promoted_to_used_keys": False,
        "entire_enumerated_source_scope_hashed_before_after": True,
        "semantic_dynamic_or_embedded_parser_input_coverage_complete": False,
        "complete_historical_usage_freshness_audit": False,
        "prior_registry_changed": False, "future_plan_accepted_or_allocated": False,
        "environment_key_draws": 0, "native_training_or_collection_started": False,
        "collection_authorized": False, "scientific_claim_authorized": False,
        "research_complete": False}
    encoded = json.dumps(payload, sort_keys=True, ensure_ascii=True, separators=(",", ":"), allow_nan=False).encode("ascii")
    return {"payload_sha256": sha(encoded), "payload": payload}


if __name__ == "__main__":
    if len(sys.argv) != 1: raise SystemExit("closed fixed scope; no output/seed/scope overrides")
    print(json.dumps(snapshot_preparation_fixtures(), ensure_ascii=True, sort_keys=True, allow_nan=False))
